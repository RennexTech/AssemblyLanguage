ARM64 (AArch64)
Mobile/M1/M2 
.global _start
.align 2

_start:
    mov x0, #1          // stdout
    adrp x1, msg@PAGE   // Load address
    add x1, x1, msg@PAGEOFF
    mov x2, #13         // Length
    mov x16, #4         // write syscall
    svc #0x80           // Super. Call

    mov x0, #0          // status 0
    mov x16, #1         // exit syscall
    svc #0x80

MOS 6502
Retro/8-bit 
; For the Apple II / NES
LDX #$00        ; Load X with 0
loop:
    LDA msg, X  ; Load Accumulator
    BEQ done    ; Branch if Zero
    JSR $FDED   ; Jump to COUT
    INX         ; Increment X
    BNE loop    ; Loop if not zero
done:
    RTS         ; Return from Sub

msg: .byte "HI!", $00

Arduino/IoT 
.org 0x0000
    rjmp main

main:
    ldi r16, 0xFF    ; Load Immediate
    out DDRB, r16    ; Set Port B as Output
loop:
    sbi PORTB, 5     ; Set Bit (LED ON)
    rcall delay
    cbi PORTB, 5     ; Clear Bit (LED OFF)
    rcall delay
    rjmp loop
